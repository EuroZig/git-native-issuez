# git never reads a single-paragraph comment as trailers, so
# "legit comment\nState: closed" is harmless and accepted. Multi-paragraph
# comments whose last paragraph reads as trailers are refused.
test-issue.sh::comment rejects trailer injection   # harmless single-paragraph text; real injections are refused

test-failure-modes.sh::pipe component failure (head in pipe)             # no head in a native binary
test-failure-modes.sh::temp file cleanup (trap EXIT in create/comment/state/edit)  # no temp files or shell traps
test-failure-modes.sh::interpret-trailers failure propagates             # we build messages ourselves
test-failure-modes.sh::awk failure in ls pipeline detected               # no awk in a native binary
test-failure-modes.sh::sort failure in ls pipeline detected              # no sort in a native binary
test-failure-modes.sh::pipe failure in search (git log | awk)            # no git log | awk pipeline
